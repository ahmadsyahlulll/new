HEADERS = {'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'}
TIMEOUT = 10
MAX_WORKERS_CHECK = 20   # buat pengecekan folder paralel
MAX_WORKERS_RECURSIVE = 5 # buat recursive scan per top-level folder (optional)

# ==================================================================
# FUNGSI DETEKSI CEPAT APAKAH SITUS WORDPRESS
# ==================================================================
def is_wordpress_site(base_url):
    """Cek apakah situs menggunakan WordPress (cepat dengan HEAD/GET beberapa file penting)."""
    test_paths = [
        "wp-login.php",
        "wp-content/",
        "wp-includes/",
        "wp-admin/"
    ]
    for path in test_paths:
        url = urljoin(base_url, path)
        try:
            resp = requests.head(url, headers=HEADERS, timeout=5, allow_redirects=False)
            if resp.status_code == 200:
                return True
            # Jika HEAD tidak support, coba GET dengan stream
            resp = requests.get(url, headers=HEADERS, timeout=5, stream=True)
            if resp.status_code == 200:
                return True
        except:
            continue
    return False

# ==================================================================
# FUNGSI SCANNING INDEX OF (PARALEL)
# ==================================================================
def is_index_of(url):
    """Cek apakah suatu URL adalah halaman Index Of (return True/False)."""
    try:
        resp = requests.get(url, headers=HEADERS, timeout=TIMEOUT, allow_redirects=False)
        if resp.status_code != 200:
            return False
        soup = BeautifulSoup(resp.text, 'html.parser')
        title = soup.find('title')
        return title and "Index of" in title.text
    except:
        return False

def get_all_open_folders_parallel(base_url, folder_list):
    """Cek banyak folder secara paralel, return list URL yang index of."""
    full_urls = [urljoin(base_url, folder) for folder in folder_list]
    open_folders = []
    with ThreadPoolExecutor(max_workers=MAX_WORKERS_CHECK) as executor:
        future_to_url = {executor.submit(is_index_of, url): url for url in full_urls}
        for future in as_completed(future_to_url):
            url = future_to_url[future]
            if future.result():
                print(f"  ✅ TERBUKA: {url}")
                open_folders.append(url)
    return open_folders

# ==================================================================
# SCANNING REKURSIF (dengan caching visited)
# ==================================================================
def get_directory_links(url):
    """Ambil daftar subdirektori dari satu halaman Index Of."""
    try:
        resp = requests.get(url, headers=HEADERS, timeout=TIMEOUT)
        if resp.status_code != 200:
            return []
        soup = BeautifulSoup(resp.text, 'html.parser')
        title = soup.find('title')
        if not (title and "Index of" in title.text):
            return []
        links = []
        for row in soup.find_all('tr'):
            link_tag = row.find('a')
            if link_tag and link_tag.get('href'):
                href = link_tag['href']
                if href.endswith('/') and href not in ('../', '/'):
                    links.append(urljoin(url, href))
        return links
    except:
        return []

def recursive_scan(folder_url, max_depth=3, visited=None, current_depth=0):
    if visited is None:
        visited = set()
    if current_depth >= max_depth or folder_url in visited:
        return []
    visited.add(folder_url)
    print(f"{'  ' * current_depth}📂 Scanning: {folder_url}")
    subdirs = get_directory_links(folder_url)
    all_sub = []
    for sub in subdirs:
        all_sub.append(sub)
        all_sub.extend(recursive_scan(sub, max_depth, visited, current_depth+1))
    return all_sub

# DETEKSI BACKDOOR (tetap pakai threading untuk file .php)
# ==================================================================
def is_suspicious_backdoor(html, status_code):
    if not html:
        return False, {}
    soup = BeautifulSoup(html, 'html.parser')
    forms = soup.find_all('form', method=lambda x: x and x.lower() == 'post')
    if not forms:
        return False, {}
    has_pwd = any(form.find_all('input', {'type': 'password'}) for form in forms)
    if not has_pwd:
        return False, {}
    title = soup.find('title')
    title_text = title.get_text(strip=True).lower() if title else ""
    keywords_title = ['pheditor', 'file manager', 'webshell', 'backdoor', 'wso', 'c99', 'b374k', 'shell']
    keywords_body = ['过安全狗', '云锁', 'd盾', 'Pheditor', 'c99shell', '微型后门', '一句话']
    fake_error = status_code == 200 and ('internal server error' in title_text or 'forbidden' in title_text)
    if any(k in title_text for k in keywords_title) or any(k in html.lower() for k in keywords_body) or fake_error:
        return True, {'title': title_text, 'fake_error': fake_error}
    return False, {}

def check_php_file(url):
    try:
        resp = requests.get(url, headers=HEADERS, timeout=10, allow_redirects=False)
        if resp.status_code in (200, 403, 404, 500):
            is_bd, reason = is_suspicious_backdoor(resp.text, resp.status_code)
            if is_bd:
                return url, True, reason
    except:
        pass
    return url, False, {}

def get_php_files_from_dir(dir_url):
    php_files = set()
    try:
        resp = requests.get(dir_url, headers=HEADERS, timeout=10)
        if resp.status_code == 200:
            soup = BeautifulSoup(resp.text, 'html.parser')
            for link in soup.find_all('a'):
                href = link.get('href')
                if href and href.endswith('.php'):
                    full = urljoin(dir_url, href)
                    php_files.add(full)
    except:
        pass
    php_files.add(urljoin(dir_url, 'index.php'))
    return php_files

def scan_backdoors_parallel(folder_urls, output_file):
    print(f"\n🔍 Mengumpulkan file .php dari {len(folder_urls)} direktori...")
    all_php = set()
    for folder in folder_urls:
        all_php.update(get_php_files_from_dir(folder))
    print(f"📄 Total file .php unik: {len(all_php)}")
    print("🚀 Deteksi backdoor (parallel)...")
    suspicious = []
    with ThreadPoolExecutor(max_workers=10) as ex:
        futures = [ex.submit(check_php_file, url) for url in all_php]
        for fut in as_completed(futures):
            url, is_bd, reason = fut.result()
            if is_bd:
                suspicious.append((url, reason))
                print(f"[!] 🚨 BACKDOOR: {url}")
    with open(output_file, 'w', encoding='utf-8') as f:
        f.write("REPORT BACKDOOR\n")
        for url, reason in suspicious:
            f.write(f"{url}\n    {reason}\n")
    print(f"✅ Ditemukan {len(suspicious)} backdoor. Laporan: {output_file}")

def process_site(base_url):
    print(f"\n{'='*60}\n🌐 Memproses: {base_url}\n{'='*60}")
    start_time = time.time()
    
    # 1. Deteksi WordPress cepat
    if not is_wordpress_site(base_url):
        print("❌ BUKAN WORDPRESS atau tidak dapat diakses. Skip.")
        return
    
    print("✅ Terdeteksi sebagai WordPress. Melanjutkan scan folder...")
    
    # 2. Scan semua folder di ALL_WP_FOLDERS secara paralel
    open_folders = get_all_open_folders_parallel(base_url, ALL_WP_FOLDERS)
    
    if not open_folders:
        print("⚠️ Tidak ada folder Index Of. Coba fallback ke file index.php...")
        fallback = [urljoin(base_url, p) for p in [
            "wp-admin/index.php", "wp-content/index.php", "wp-includes/index.php",
            "wp-content/plugins/index.php", "wp-includes/js/index.php"
        ]]
        scan_backdoors_parallel(fallback, f"backdoor_{base_url.replace('https://','').replace('/','_')}.txt")
        return
    
    # 3. Filter top-level (parent tidak terbuka)
    open_set = set(open_folders)
    top_level = []
    for folder in sorted(open_folders, key=len):
        parent = folder.rstrip('/')
        last = parent.rfind('/')
        if last != -1:
            parent = parent[:last+1]
            if parent in open_set:
                continue
        top_level.append(folder)
    
    print(f"\n🎯 Folder top-level ({len(top_level)}) akan discan rekursif:")
    for f in top_level:
        print(f"   {f}")
    
    # 4. Scan rekursif (bisa sequential karena sudah terfilter, cepat)
    all_dirs = []
    for folder in top_level:
        print(f"\n📁 Rekursif: {folder}")
        subdirs = recursive_scan(folder, max_depth=3)
        all_dirs.append(folder)
        all_dirs.extend(subdirs)
    
    all_dirs = list(dict.fromkeys(all_dirs))
    print(f"\n✨ Total direktori unik: {len(all_dirs)}")
    
    # 5. Scan backdoor di semua direktori
    output_file = f"backdoor_{base_url.replace('https://','').replace('http://','').replace('/','_')}.txt"
    scan_backdoors_parallel(all_dirs, output_file)
    
    elapsed = time.time() - start_time
    print(f"⏱️  Waktu total untuk {base_url}: {elapsed:.2f} detik")

# MAIN: BATCH DARI FILE ATAU SINGLE URL
# ==================================================================
def main():
    if len(sys.argv) < 2:
        print("Usage: python main.py <url_or_file.txt>")
        print("Contoh: python main.py https://example.com")
        print("        python main.py daftar_situs.txt")
        sys.exit(1)
    
    arg = sys.argv[1]
    if arg.endswith('.txt'):
        with open(arg, 'r') as f:
            urls = [line.strip() for line in f if line.strip()]
        for url in urls:
            if not url.startswith(('http://','https://')):
                url = 'https://' + url
            process_site(url)
    else:
        if not arg.startswith(('http://','https://')):
            arg = 'https://' + arg
        process_site(arg)

if __name__ == "__main__":
    main()
